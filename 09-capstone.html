python · onto.py
CLASSES = {"Thing": None, "Human": "Thing", "Project": "Thing",
           "Technique": "Thing", "Paper": "Thing", "Tool": "Thing"}
RELATIONS = {"leads": ("Human", "Project"),
             "belongs_to": ("Project", "Project"),
             "uses": ("Project", "Technique"),
             "cites": ("Paper", "Paper"),
             "supports": ("Paper", "Technique"),
             "implemented_by": ("Technique", "Tool")}

class Ontology:
    def check(self, rel, subj, obj):
        """合法返回 True；否则返回人类可读的错误说明（会喂回模型/展示给用户）"""
        if rel not in RELATIONS:
            return f"未知关系 {rel}，可用: {sorted(RELATIONS)}"
        d, r = RELATIONS[rel]
        # 轻量版只拦「明确冲突」；实体类型由注册表在写入时登记
        self._registry = getattr(self, "_registry", {})
        for name, cls in ((subj, d), (obj, r)):
            known = self._registry.get(name)
            if known and known not in _ancestors(known, CLASSES) | {known}:
                return f"{name} 的类型 {known} 与 {rel} 的要求不符"
        self._registry.setdefault(subj, d)
        self._registry.setdefault(obj, r)
        return True

def _ancestors(cls, classes):
    out = set()
    c = classes.get(cls)
    while c:
        out.add(c); c = classes.get(c)
    return out
python · memory.py
import json, pathlib

class TripleStore:
    def __init__(self, path="kg.json"):
        self.path = pathlib.Path(path)
        self.triples = (json.loads(self.path.read_text("utf-8"))
                        if self.path.exists() else [])

    def add(self, rel, subj, obj):
        t = (rel, subj, obj)
        if t not in self.triples:
            self.triples.append(t)
            self.path.write_text(
                json.dumps(self.triples, ensure_ascii=False, indent=1), "utf-8")
        return {"ok": True, "stored": f"{subj} -{rel}-> {obj}"}

    def query(self, rel=None, subj=None, obj=None):
        hits = [t for t in self.triples
                if (rel is None or t[0] == rel)
                and (subj is None or t[1] == subj)
                and (obj is None or t[2] == obj)]
        return {"count": len(hits),
                "triples": [f"{s} -{r}-> {o}" for r, s, o in hits]}

    def summary(self):
        return {"triples": [f"{s} -{r}-> {o}" for r, s, o in self.triples]}

from onto import Ontology
from memory import TripleStore

def test_unknown_relation_rejected():
    assert Ontology().check("loves", "Alice", "猫") != True

def test_wrong_domain_detected():
    onto = Ontology()
    onto.check("leads", "Alice", "P1")            # Alice 登记为 Human
    assert Ontology().check("uses", "Alice", "重排序") != True  # Human 不能 uses

def test_add_and_query():
    s = TripleStore(":memory:")                   # 测试用独立路径
    s.add("leads", "Alice", "GraphRAG对比实验")
    assert s.query(subj="Alice")["count"] == 1
    assert s.query(rel="cites")["count"] == 0

def test_dedup():
    s = TripleStore(":memory:")
    s.add("leads", "Alice", "P1"); s.add("leads", "Alice", "P1")
    assert len(s.triples) == 1

def test_persistence(tmp_path):
    p = tmp_path / "kg.json"
    TripleStore(p).add("leads", "A", "B")
    assert TripleStore(p).query(subj="A")["count"] == 1

python · registry.py
class OntoRegistry:
    def __init__(self, onto, store):
        self.onto = onto; self.store = store; self._tools = {}

    def register(self, spec, impl, permission="auto", category="general"):
        self._tools[spec["function"]["name"]] = (spec, impl, permission, category)

    def specs_for(self, categories=None):
        return [t[0] for t in self._tools.values()
                if categories is None or t[3] in categories]

    def call(self, name, args, confirm_fn):
        if name not in self._tools:
            return {"error": f"未知工具 {name}，可用: {sorted(self._tools)}"}
        _, impl, perm, cat = self._tools[name]
        if perm == "confirm" and not confirm_fn(f"执行 {name}({args})?"):
            return {"error": "用户拒绝执行该操作。"}
        if cat == "memory-write":                      # 本体校验作为闸门
            ok = self.onto.check(args.get("rel"),
                                 args.get("subj"), args.get("obj"))
            if ok is not True:
                return {"error": ok}
        try:
            return impl(**args)
        except Exception as e:
            return {"error": f"{type(e).__name__}: {e}"}
python · harness.py
import json, time
from openai import OpenAI

client = OpenAI(api_key=os.environ["DEEPSEEK_API_KEY"],
                base_url="https://api.deepseek.com")

class Trace:
    def __init__(self, path="trace.jsonl"): self.path = path; self.t0 = time.time()
    def log(self, kind, **kw):
        rec = {"t": round(time.time() - self.t0, 2), "kind": kind, **kw}
        with open(self.path, "a", encoding="utf-8") as f:
            f.write(json.dumps(rec, ensure_ascii=False) + "\\n")

class Harness:
    def __init__(self, registry, system_prompt, max_steps=10):
        self.reg = registry; self.sys = system_prompt
        self.max_steps = max_steps; self.trace = Trace()

    def confirm(self, question):                    # CLI 版；M4 换成 Web 审批
        return input(f"  {question} [y/N] ").strip().lower() == "y"

    def run(self, user_input):
        messages = [{"role": "system", "content": self.sys},
                    {"role": "user", "content": user_input}]
        for step in range(self.max_steps):
            resp = client.chat.completions.create(
                model="deepseek-chat", messages=messages,
                tools=self.reg.specs_for(), temperature=0)
            self.trace.log("llm", step=step, tokens=resp.usage.total_tokens)
            msg = resp.choices[0].message
            if not msg.tool_calls:
                self.trace.log("final", step=step)
                return msg.content
            messages.append(msg)
            for tc in msg.tool_calls:
                args = json.loads(tc.function.arguments or "{}")
                self.trace.log("tool", step=step, name=tc.function.name, args=args)
                result = self.reg.call(tc.function.name, args, self.confirm)
                messages.append({"role": "tool", "tool_call_id": tc.id,
                                 "content": json.dumps(result, ensure_ascii=False)})
        return "已达最大步数，任务未完成。"

    {"rel": {"type": "string", "description": "必须是: " + "/".join(RELATIONS)},

reg.register(f("query_graph",
    "按条件查询知识图谱。回答涉及用户过往信息前必须先查图，禁止凭空回忆。",
    {"rel": {"type": "string"}, "subj": {"type": "string"}, "obj": {"type": "string"}}, []),

    category="memory-read")

reg.register(f("graph_summary",
    "返回整个知识图谱的内容。用户问『我都记了什么』时使用。",
    {}, []),
    impl=lambda: store.summary(), category="memory-read")

    "从一段文本（论文摘要/笔记）中自动抽取三元组并入图。",

M2 验收标准（三个场景脚本）

① 说「记住我的课题叫 GraphRAG对比实验」→ 回复包含「已记录」且图里多一条 leads；
② 问「我的课题叫什么」→ 模型先调 query_graph 再回答（看 Trace 确认）；
③ 调 clear_all_memory → 弹出确认，回答 N → 工具返回「用户拒绝」且图谱无损、模型体面收场。

常见坑（往届「毕业生」的血泪）

  • 模型自说自话不查图：系统提示里「先查图再回答」要写成硬规则，并配一条 evals 用例盯着（看 Trace 有没有 query_graph）
  • 关系枚举漂移：模型发明新关系「研究」——schema 描述里把枚举写死，校验兜底
  • 实体名不统一：图谱里出现「GraphRAG对比实验」「我的课题」两个节点——抽取提示词里带上已有实体清单
  • 确认流形同虚设：Web 版把 confirm 做成了「自动通过」——验收时专测这一条
  • 只测晴天：评估集全是顺利场景——边界层的 5 条一条都不能少
